### Task 2: Build full-width search section

**Objective:** Add search bar at top of Browse page with instant results grid.

**Files:**
- Create: `src/components/BrowseSearch.jsx`
- Modify: `src/pages/Browse.jsx`
- Modify: `src/styles/globals.css`

**Step 1: Create BrowseSearch.jsx**

```jsx
import { useState, useEffect } from 'react'
import { Link } from 'react-router'
import { searchBooks } from '../lib/openlibrary'
import BookCard from './BookCard'

const QUICK_FILTERS = ['Fiction', 'Non-Fiction', 'Science', 'History', 'Fantasy', 'Romance']

export default function BrowseSearch({ onSearch }) {
  const [query, setQuery] = useState('')
  const [results, setResults] = useState([])
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    if (!query.trim()) {
      setResults([])
      return
    }
    const timer = setTimeout(async () => {
      setLoading(true)
      const books = await searchBooks(query)
      setResults(books)
      setLoading(false)
    }, 300)
    return () => clearTimeout(timer)
  }, [query])

  return (
    <section className="browse-search">
      <div className="browse-search-bar">
        <input
          className="browse-search-input"
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search for books, authors, or topics..."
          type="text"
          value={query}
        />
      </div>
      <div className="browse-filters">
        {QUICK_FILTERS.map((filter) => (
          <button
            className="browse-filter-chip"
            key={filter}
            onClick={() => setQuery(filter)}
            type="button"
          >
            {filter}
          </button>
        ))}
      </div>
      {query.trim() && (
        <div className="browse-results">
          {loading && <p className="browse-loading">Searching...</p>}
          {!loading && results.length === 0 && (
            <p className="browse-empty">No books found for "{query}"</p>
          )}
          {!loading && results.length > 0 && (
            <div className="browse-results-grid">
              {results.map((book) => (
                <BookCard book={book} key={book.key} />
              ))}
            </div>
          )}
        </div>
      )}
    </section>
  )
}
```

**Step 1: Create BrowseGenre.jsx**

```jsx
import { useState } from 'react'
import { fetchSubjectBooks } from '../lib/openlibrary'
import BookCard from './BookCard'

const GENRES = [
  { id: 'fiction', label: 'Fiction', color: 'var(--cover-sage)' },
  { id: 'science_fiction', label: 'Sci-Fi', color: 'var(--cover-terracotta)' },
  { id: 'fantasy', label: 'Fantasy', color: 'var(--cover-olive)' },
  { id: 'romance', label: 'Romance', color: 'var(--cover-umber)' },
  { id: 'mystery', label: 'Mystery', color: 'var(--brand-accent)' },
  { id: 'history', label: 'History', color: 'var(--brand-primary)' },
  { id: 'science', label: 'Science', color: 'var(--cover-sage-deep)' },
  { id: 'poetry', label: 'Poetry', color: 'var(--cover-terracotta)' },
]

export default function BrowseGenre() {
  const [expanded, setExpanded] = useState(null)
  const [books, setBooks] = useState([])
  const [loading, setLoading] = useState(false)

  const handleGenreClick = async (genreId) => {
    if (expanded === genreId) {
      setExpanded(null)
      return
    }
    setExpanded(genreId)
    setLoading(true)
    const fetched = await fetchSubjectBooks(genreId)
    setBooks(fetched)
    setLoading(false)
  }

  return (
    <section className="browse-genres">
      <h2 className="browse-section-title">Browse by Genre</h2>
      <div className="browse-genre-grid">
        {GENRES.map((genre) => (
          <div
            className={`browse-genre-card${expanded === genre.id ? ' browse-genre-card--expanded' : ''}`}
            key={genre.id}
          >
            <button
              className="browse-genre-header"
              onClick={() => handleGenreClick(genre.id)}
              type="button"
              style={{ '--genre-color': genre.color }}
            >
              <span className="browse-genre-label">{genre.label}</span>
              <span className="browse-genre-arrow">{expanded === genre.id ? '−' : '+'}</span>
            </button>
            {expanded === genre.id && (
              <div className="browse-genre-content">
                {loading && <p className="browse-loading">Loading books...</p>}
                {!loading && books.length > 0 && (
                  <div className="browse-genre-books">
                    {books.slice(0, 5).map((book) => (
                      <BookCard book={book} key={book.key} />
                    ))}
                  </div>
                )}
              </div>
            )}
          </div>
        ))}
      </div>
    </section>
  )
}
```

**Step 1: Create NewArrivals.jsx**

```jsx
import { useEffect, useState } from 'react'
import BookCard from './BookCard'

export default function NewArrivals() {
  const [books, setBooks] = useState([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    async function fetchNew() {
      try {
        const res = await fetch(
          'https://openlibrary.org/search.json?sort=new&limit=10&fields=key,title,author_name,first_publish_year,cover_i,rating_average'
        )
        const data = await res.json()
        setBooks(data.docs || [])
      } catch {
        setBooks([])
      }
      setLoading(false)
    }
    fetchNew()
  }, [])

  if (loading) return <p className="browse-loading">Loading new arrivals...</p>
  if (books.length === 0) return null

  return (
    <section className="browse-section">
      <h2 className="browse-section-title">New Arrivals</h2>
      <div className="browse-scroll-row">
        {books.map((book) => (
          <div className="browse-scroll-item" key={book.key}>
            <BookCard book={book} />
          </div>
        ))}
      </div>
    </section>
  )
}
```
